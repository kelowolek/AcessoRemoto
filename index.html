Enviando autenticação..."); ws.send(JSON.stringify({ type: 'AUTH', deviceId: document.getElementById('deviceId').value, password: document.getElementById('accessPassword').value })); }; // restante da sua lógica de recebimento das imagens (ws.onmessage)... }
